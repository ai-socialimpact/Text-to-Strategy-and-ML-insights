-- (1) COHORT: Batch 14, FEMALE, exact category 'Medical' (excludes Pharmacy;
--     Educational_Stream 'Medical / Public Health' is deliberately NOT used)
CREATE OR REPLACE VIEW v_cohort AS
SELECT DISTINCT
       TRIM(CAST(Participant_Id AS VARCHAR)) AS participant_id
FROM   prod_nirman_demographics
WHERE  TRY_CAST(Workshop_Batch_Number AS DOUBLE) = 14
  AND  TRIM(Gender) = 'FEMALE'
  AND  TRIM(Educational_Stream_Category) = 'Medical';


-- (2) PRE POINT-1 ROWS: semi-join via IN (never multiplies cohort rows)
CREATE OR REPLACE VIEW v_pre_rows AS
SELECT q.file_row_id,
       TRIM(CAST(q.Participant_Id AS VARCHAR))              AS participant_id,
       CAST(q.Metric_Life_Satisfaction_Global AS VARCHAR)   AS raw_txt
FROM   prod_nirman_questionnaire AS q
WHERE  TRIM(CAST(q.Workshop_Phase AS VARCHAR)) = 'Pre'
  AND  ABS(TRY_CAST(q.Workshop_Level AS DOUBLE) - 0.1) < 1e-9
  AND  TRIM(CAST(q.Participant_Id AS VARCHAR)) IN (SELECT participant_id FROM v_cohort);


-- Parse the score and flag validity (integer 1–5)
CREATE OR REPLACE VIEW v_pre_scored AS
SELECT r.*,
       TRY_CAST(NULLIF(TRIM(raw_txt), '') AS DOUBLE) AS v,
       CASE WHEN TRY_CAST(NULLIF(TRIM(raw_txt), '') AS DOUBLE) IS NOT NULL
             AND TRY_CAST(NULLIF(TRIM(raw_txt), '') AS DOUBLE)
                 = FLOOR(TRY_CAST(NULLIF(TRIM(raw_txt), '') AS DOUBLE))
             AND TRY_CAST(NULLIF(TRIM(raw_txt), '') AS DOUBLE) BETWEEN 1 AND 5
            THEN 1 ELSE 0 END AS is_valid
FROM   v_pre_rows r;


-- (4) KEEP-ONE RULE: first valid row in file order, else first row (flagged invalid)
CREATE OR REPLACE VIEW v_pre_dedup AS
SELECT participant_id, file_row_id, raw_txt, v, is_valid
FROM (
    SELECT s.*,
           ROW_NUMBER() OVER (
               PARTITION BY participant_id
               ORDER BY CASE WHEN is_valid = 1 THEN 0 ELSE 1 END,   -- valid rows first
                        file_row_id                                  -- then file order
           ) AS rn
    FROM v_pre_scored s
) t
WHERE rn = 1;


-- (5) QUALITY BUCKETS: mutually exclusive, evaluated in the specified order
CREATE OR REPLACE VIEW v_quality AS
SELECT participant_id, v,
       CASE
         WHEN raw_txt IS NULL OR TRIM(raw_txt) = '' OR LOWER(TRIM(raw_txt)) = 'nan'
                                                  THEN 'blank_nan_null'
         WHEN v IS NULL                           THEN 'non_numeric'
         WHEN v = 0                               THEN 'zero'
         WHEN v > 5                               THEN 'above_5'
         WHEN v = FLOOR(v) AND v BETWEEN 1 AND 5  THEN 'valid'
         ELSE                                          'other_diagnostic'
       END AS bucket
FROM   v_pre_dedup;

-- Order-statistic median, written out explicitly (x[(N-1)/2] or average of two middle values)
WITH ordered AS (

           ROW_NUMBER() OVER (ORDER BY v) AS rn,
           COUNT(*)     OVER ()           AS n
    FROM   v_quality
    WHERE  bucket = 'valid'

Method 3: Frequency-table weighted reconstruction

Everything is derived from the 5-row frequency vector, which is natural in SQL.

sql
WITH freq AS (
    SELECT k.score, COUNT(q.v) AS f
    FROM  (VALUES (1),(2),(3),(4),(5)) AS k(score)
    LEFT JOIN v_quality q ON q.bucket = 'valid' AND q.v = k.score
    GROUP BY k.score
),
tot AS (
    SELECT SUM(f) AS n, SUM(score * f) AS weighted_sum FROM freq
),
cum AS (
    SELECT score, f,
           SUM(f) OVER (ORDER BY score)       AS cum_count,
           SUM(f) OVER (ORDER BY score) - f   AS prev_cum
    FROM   freq
),
median_positions AS (                       -- positions ceil(N/2) and floor(N/2)+1
    SELECT CEIL(n / 2.0)      AS pos FROM tot
    UNION ALL
    SELECT FLOOR(n / 2.0) + 1 AS pos FROM tot
),
median_calc AS (
    SELECT AVG(c.score * 1.0) AS median_val
    FROM   median_positions p
    JOIN   cum c ON p.pos > c.prev_cum AND p.pos <= c.cum_count
)
SELECT t.n                                                       AS valid_n,
       ROUND(1.0 * t.weighted_sum / NULLIF(t.n, 0), 2)           AS weighted_mean,
       (SELECT median_val FROM median_calc)                      AS median,
       (SELECT SUM(f) FROM freq WHERE score >= 4)                AS agree_count,
       ROUND(100.0 * (SELECT SUM(f) FROM freq WHERE score >= 4) / NULLIF(t.n, 0), 2) AS agree_pct,
       ROUND(100.0 * (SELECT SUM(f) FROM freq WHERE score >= 4) / 52, 2)              AS agree_pct_of_cohort,
       -- assertion: sum of frequencies = valid N (compared with an independent count)
       t.n = (SELECT COUNT(*) FROM v_quality WHERE bucket = 'valid') AS assert_sum_freq_eq_valid_n,
       -- cross-check: weighted mean vs raw mean (|diff| < 1e-9)
       ABS(1.0 * t.weighted_sum / NULLIF(t.n, 0)
           - (SELECT AVG(v) FROM v_quality WHERE bucket = 'valid')) < 1e-9 AS assert_mean_crosscheck
FROM tot t;

-- Weight shares f_k / N × 100
SELECT score, f,
       ROUND(100.0 * f / NULLIF(SUM(f) OVER (), 0), 2) AS share_pct
FROM   freq
ORDER  BY score;

-- (b) Wilson 95% CI for agree proportion (z = 1.96)
WITH s AS (
    SELECT COUNT(*)                                   AS n,
           COUNT(*) FILTER (WHERE v >= 4)             AS a,
           1.96                                       AS z
    FROM   v_quality
    WHERE  bucket = 'valid'
),
p AS (SELECT n, a, z, 1.0 * a / n AS p FROM s WHERE n > 0)      -- guards N = 0
SELECT ROUND(100 * ((p + z*z/(2.0*n)) / (1 + z*z/n)
              - z * SQRT(p*(1-p)/n + z*z/(4.0*n*n)) / (1 + z*z/n)), 2) AS wilson_lower_pct,
       ROUND(100 * ((p + z*z/(2.0*n)) / (1 + z*z/n)
              + z * SQRT(p*(1-p)/n + z*z/(4.0*n*n)) / (1 + z*z/n)), 2) AS wilson_upper_pct
FROM   p;

-- (c) Sensitivity to the keep-one rule: primary vs alt A (LAST row) vs alt B (FIRST row, any validity)
WITH rules AS (
    SELECT 'primary' AS rule, participant_id, v, is_valid FROM v_pre_dedup
    UNION ALL
    SELECT 'A_keep_last', participant_id, v, is_valid
    FROM (SELECT s.*, ROW_NUMBER() OVER (PARTITION BY participant_id
                                         ORDER BY file_row_id DESC) AS rn
          FROM v_pre_scored s) x
    WHERE rn = 1
    UNION ALL
    SELECT 'B_first_any', participant_id, v, is_valid
    FROM (SELECT s.*, ROW_NUMBER() OVER (PARTITION BY participant_id
                                         ORDER BY file_row_id ASC) AS rn
          FROM v_pre_scored s) x
    WHERE rn = 1
),
agg AS (
    SELECT rule,
           COUNT(*) FILTER (WHERE is_valid = 1)                         AS n_valid,
           AVG(CASE WHEN is_valid = 1 THEN v END)                       AS mean_v,
           100.0 * COUNT(*) FILTER (WHERE is_valid = 1 AND v >= 4)
                 / NULLIF(COUNT(*) FILTER (WHERE is_valid = 1), 0)      AS agree_pct
    FROM   rules
    GROUP  BY rule
)
SELECT rule,
       n_valid,
       ROUND(mean_v, 2)                                                  AS mean,
       ROUND(agree_pct, 2)                                               AS agree_pct,
       n_valid   - FIRST_VALUE(n_valid)   OVER (ORDER BY CASE WHEN rule='primary' THEN 0 ELSE 1 END) AS delta_n,
       ROUND(mean_v    - FIRST_VALUE(mean_v)    OVER (ORDER BY CASE WHEN rule='primary' THEN 0 ELSE 1 END), 4) AS delta_mean,
       ROUND(agree_pct - FIRST_VALUE(agree_pct) OVER (ORDER BY CASE WHEN rule='primary' THEN 0 ELSE 1 END), 4) AS delta_agree_pct
FROM   agg
ORDER  BY rule;

-- Rank-based median, upper-tail agreement, mode and median percentile rank
WITH ranked AS (
    SELECT v, ROW_NUMBER() OVER (ORDER BY v) AS rnk, COUNT(*) OVER () AS n
    FROM   v_quality WHERE bucket = 'valid'
),
med AS (
    SELECT AVG(v) AS median_val, MAX(n) AS n
    FROM   ranked
    WHERE  rnk IN (FLOOR((n + 1) / 2.0), CEIL((n + 1) / 2.0))
),
f3 AS (
    SELECT COUNT(*) FILTER (WHERE v <= 3) AS cum3,
           COUNT(*)                       AS n
    FROM   v_quality WHERE bucket = 'valid'
)
SELECT m.n                                                         AS valid_n,
       ROUND(m.median_val, 2)                                      AS median,
       f3.n - f3.cum3                                              AS agree_count,          -- N − cumcount(3)
       ROUND(100.0 * (1 - 1.0 * f3.cum3 / f3.n), 2)                AS agree_pct_via_1_minus_F3,
       ROUND(100.0 * (f3.n - f3.cum3) / f3.n, 2)                   AS agree_pct_crosscheck,
       ROUND(100.0 * (f3.n - f3.cum3) / 52, 2)                     AS agree_pct_of_cohort,
       -- mid-rank percentile of the median value
       ROUND(100.0 * ( (SELECT COUNT(*) FROM v_quality WHERE bucket='valid' AND v <  m.median_val)
                     + 0.5 * (SELECT COUNT(*) FROM v_quality WHERE bucket='valid' AND v = m.median_val)
                     ) / m.n, 2)                                   AS median_percentile_rank
FROM   med m CROSS JOIN f3;

-- Modal score and its share
SELECT v AS modal_score, COUNT(*) AS f,
       ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS share_pct
FROM   v_quality
WHERE  bucket = 'valid'
GROUP  BY v
ORDER  BY COUNT(*) DESC, v
LIMIT  1;
